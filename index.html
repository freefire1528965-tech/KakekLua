// ========================================== // KAKEKLUAVIP - SCRIPT CHAT & AI LOGIC // ========================================== // 1. Fungsi Deteksi Mode Pesan (Filter Trigger) function getMessageMode(message) { const text = String(message || "").trim(); const match = text.match(/^map\s*:\s*(.*)$/i); if (match) { const mapName = match[1].trim(); if (!mapName) { return { mode: "MAP_EMPTY", mapName: "" }; } return { mode: "MAP_SCRIPT", mapName }; } return { mode: "CHAT", message: text }; } // 2. Fungsi untuk Menampilkan Pesan ke UI Chat (Bubble Chat) function appendMessage(sender, text) { const chatContainer = document.getElementById("chat-container") || document.getElementById("chatBox") || document.body; const messageDiv = document.createElement("div"); messageDiv.className = `message ${sender}-message`; messageDiv.style.margin = "8px 0"; messageDiv.style.padding = "10px 14px"; messageDiv.style.borderRadius = "8px"; messageDiv.style.maxWidth = "80%"; if (sender === "user") { messageDiv.style.background = "#007bff"; messageDiv.style.color = "#fff"; messageDiv.style.marginLeft = "auto"; } else { messageDiv.style.background = "#e9ecef"; messageDiv.style.color = "#333"; messageDiv.style.marginRight = "auto"; } messageDiv.innerText = text; chatContainer.appendChild(messageDiv); chatContainer.scrollTop = chatContainer.scrollHeight; } // 3. Fungsi Utama Menghubungkan ke API AI Anda async function callAIApi(promptText) { try { // Ganti URL atau Endpoint API AI Anda yang sudah ada di sini jika berbeda const response = await fetch("https://api.openai.com/v1/chat/completions", { method: "POST", headers: { "Content-Type": "application/json", // "Authorization": "Bearer API_KEY_ANDA" // Masukkan API key jika diperlukan }, body: JSON.stringify({ model: "gpt-3.5-turbo", messages: [{ role: "user", content: promptText }] }) }); const data = await response.json(); // Sesuaikan dengan struktur JSON balik dari API Anda const reply = data.choices ? data.choices[0].message.content : (data.reply || "Halo! Ada yang bisa Kakek bantu?"); appendMessage("bot", reply); } catch (error) { console.error("API Error:", error); appendMessage("bot", "Waduh, koneksi AI lagi sedikit kendala nih bro. Coba lagi ya!"); } } // 4. Handler Tombol Kirim / Enter async function handleSendMessage() { const inputElement = document.getElementById("user-input") || document.getElementById("chatInput"); if (!inputElement) return; const rawText = inputElement.value; if (!rawText.trim()) return; // Tampilkan pesan user ke layar appendMessage("user", rawText); // Kosongkan kolom input seketika inputElement.value = ""; // Evaluasi menggunakan aturan getMessageMode const evaluated = getMessageMode(rawText); if (evaluated.mode === "MAP_EMPTY") { appendMessage("bot", "Tolong sebutkan nama map-nya ya, bro! Contoh: map: +1 Strength Grow Arms"); return; } let finalPrompt = ""; if (evaluated.mode === "MAP_SCRIPT") { finalPrompt = `Buatkan skrip Roblox Lua khusus untuk map: ${evaluated.mapName}`; } else { // Chat normal (bebas ngobrol, tidak otomatis bikin script) finalPrompt = evaluated.message; } // Kirim ke AI await callAIApi(finalPrompt); } // 5. Otomatis pasang event listener saat halaman dimuat document.addEventListener("DOMContentLoaded", () => { const sendButton = document.getElementById("send-btn") || document.getElementById("sendButton"); const inputElement = document.getElementById("user-input") || document.getElementById("chatInput"); if (sendButton && inputElement) { sendButton.addEventListener("click", handleSendMessage); inputElement.addEventListener("keypress", (e) => { if (e.key === "Enter") { handleSendMessage(); } }); } });
