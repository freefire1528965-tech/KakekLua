async function processAIResponse(rawUserMessage) { const text = String(rawUserMessage || "").trim(); const lowerText = text.toLowerCase(); chatHistory.push(text); await new Promise(resolve => setTimeout(resolve, 700)); const mapTriggerMatch = text.match(/^map\s*:\s*(.*)$/i); if (mapTriggerMatch) { let rawQuery = mapTriggerMatch[1].trim(); if (!rawQuery) { return "Formatnya kurang lengkap, bro. Contoh: map: race for egg"; } let cleanMapName = rawQuery .replace(/(ada gak\?*|ada gak|tolong carikan|saya butuh.*|butuh script.*|cari script.*)/gi, "") .trim(); if (!cleanMapName) { cleanMapName = rawQuery; } const foundScript = searchScriptFromDatabase(cleanMapName); if (foundScript) { return `// KakekLua Script Hub [DITEMUKAN]\n-- Target Map: ${cleanMapName}\n\n${foundScript}`; } else { return `Waduh bro, untuk map "${cleanMapName}" script-nya belum ketemu nih di database kita. Coba cek kata kunci atau nama map-nya lagi ya!`; } } if (lowerText.match(/^(halo|hai|p|assalamu[']?alaikum|hi)/)) { return "Halo bro 😎"; } if (lowerText.includes("apa kabar") || lowerText.includes("gimana kabarmu")) { return "Baik bro 😎 lu sendiri gimana?"; } if (lowerText.includes("cari bantuan") || lowerText.includes("minta tolong") || lowerText.includes("butuh bantuan")) { return "Siap bro, bantuan soal apa?"; } if (lowerText.includes("cari script") || lowerText.includes("butuh script") || lowerText.includes("mau script")) { return "Siap bro 😎 script apa yang lu cari?"; } if (lowerText.includes("error") || lowerText.includes("bug") || lowerText.includes("masalah")) { return "Errornya di bagian mana bro?"; } if (lowerText.includes("bingung") || lowerText.includes("pusing") || lowerText.includes("ribet")) { return "Wkwk kenapa bro? Cerita aja."; } if (lowerText.includes("lagi ngapain") || lowerText.includes("lagi sibuk") || lowerText.includes("sibuk apa")) { return "Lagi standby nih wkwk 😎"; } const previousMessage = chatHistory.length > 1 ? chatHistory[chatHistory.length - 2].toLowerCase() : ""; if (previousMessage.includes("cari script") || previousMessage.includes("script apa")) { return `Oke bro 😎 untuk skrip ${text}, nanti tinggal ketik map: [NamaMap] kalau mau langsung dicariin datanya ya!`; } return `Oh ngomongin soal "${text}", ada hal spesifik lain yang mau dibahas atau ditanyakan, bro?`; } function searchScriptFromDatabase(mapName) { const databaseSimulasi = { "race for egg": `local plrs = game.Players.LocalPlayer\nprint("Auto Farm Active for Race For Egg!")\n-- Sumber: Database KakekLua`, }; const key = mapName.toLowerCase(); if (databaseSimulasi[key]) { return databaseSimulasi[key]; } return null; }
