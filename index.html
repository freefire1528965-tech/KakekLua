function handleAIResponse(userText) { const text = userText.toLowerCase().trim(); // 1. Deteksi kalau cuma ngobrol / basa-basi / sapaan const chatKeywords = ["halo", "hai", "p", "wkwk", "woy", "assalamualaikum", "test", "pagi", "malam", "min", "bang", "bos"]; const isOnlyChat = chatKeywords.some(word => text === word || text.startsWith(word + " ")); // Kalau teksnya mengandung unsur ngobrol santai / curhat singkat if (isOnlyChat || text.length < 5 && !isGameName(text)) { return { type: "chat", text: "Wkwk santai bro! Mau ngobrol dulu apa mau cari script game apa nih hari ini? Ketik aja nama gamenya kalau udah siap ya!" }; } // 2. Kalau beneran mau cari script game let scriptResult = searchInDatabase(text); if (scriptResult) { return { type: "script", gameName: userText, loadstring: scriptResult.url }; } else { return { type: "chat", text: `Duh, kayaknya script buat "${userText}" belum ada deh. Coba ketik nama game lain yang bener wkwk!` }; } }
