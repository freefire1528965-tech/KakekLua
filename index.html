import os import aiohttp import discord from discord.ext import commands # Konfigurasi Bot intents = discord.Intents.default() intents.message_content = True bot = commands.Bot(command_prefix="!", intents=intents) @bot.event async def on_ready(): print(f'Asisten Mandiri {bot.user.name} sudah online dan siap mencari script!') # Fungsi pencarian otomatis ke API ScriptBlox async def search_scriptblox(query): url = f"https://scriptblox.com/api/script/search?q={query}&mode=free" async with aiohttp.ClientSession() as session: async with session.get(url) as response: if response.status == 200: data = await response.json() results = data.get("result", {}).get("scripts", []) return results[:3] # Ambil 3 script teratas yang paling baru return [] @bot.command(name="cariscrip", help="Cari script otomatis. Contoh: !cariscrip Race for Eggs") async def cariscrip(ctx, *, game_name: str): await ctx.send(f"🔍 Sedang menjelajahi web & database hacker untuk game: **{game_name}**...") scripts = await search_scriptblox(game_name) if not scripts: await ctx.send("❌ Waduh, script terbaru belum ditemukan atau nama gamenya kurang spesifik.") return embed = discord.Embed(title=f"📦 Hasil Pencarian Script: {game_name}", color=0x00FF00) for i, s in enumerate(scripts, 1): title = s.get("title", "No Title") verified = "✅ Verified" if s.get("verified") else "⚠️ Unverified" game_title = s.get("game", {}).get("name", "Unknown Game") script_code = s.get("script", "loadstring(...)()") # Batasi panjang kode jika terlalu panjang untuk embed if len(script_code) > 1000: script_code = script_code[:1000] + "\n... (kode dipotong karena terlalu panjang, salin dari source)" embed.add_field( name=f"{i}. {title} ({verified})", value=f"**Game:** {game_title}\n```lua\n{script_code}\n```", inline=False ) await ctx.send(embed=embed) # Jalankan bot (Masukkan token bot Discord kamu di sini) # bot.run("TOKEN_DISCORD_KAMU_DISINI")
