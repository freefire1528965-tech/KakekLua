// ========================================== // 1. FUNGSI DETEKSI MODE PESAN // ========================================== function getMessageMode(message) { const text = String(message || "").trim(); // Hanya "map:" di awal pesan yang menjadi trigger khusus const match = text.match(/^map\s*:\s*(.*)$/i); if (match) { const mapName = match[1].trim(); if (!mapName) { return { mode: "MAP_EMPTY", mapName: "" }; } return { mode: "MAP_SCRIPT", mapName }; } // Semua pesan lainnya = percakapan normal AI return { mode: "CHAT", message: text }; } // ========================================== // 2. INTEGRASI KE EVENT KIRIM PESAN // ========================================== // Panggil fungsi ini saat user menekan tombol Kirim / Enter async function handleUserSend(rawInputText) { try { if (!rawInputText || !rawInputText.trim()) return; // Evaluasi pesan menggunakan fungsi pembatas trigger const evaluated = getMessageMode(rawInputText); let payloadToSend = ""; if (evaluated.mode === "MAP_EMPTY") { // Jika user hanya mengetik "map:" tanpa nama map if (typeof appendBotMessageToUI === "function") { appendBotMessageToUI("Tolong sebutkan nama map-nya ya, bro! Contoh: map: +1 Strength Grow Arms"); } return; } else if (evaluated.mode === "MAP_SCRIPT") { // Masuk mode script khusus map payloadToSend = `Buatkan script Roblox Lua untuk map: ${evaluated.mapName}`; } else { // Chat normal (mencegah bot langsung ngide bikin script kalau cuma ngechat biasa) payloadToSend = evaluated.message; } // Lanjutkan ke fungsi API AI yang sudah ada di project kamu if (typeof sendToExistingAPI === "function") { await sendToExistingAPI(payloadToSend); } } catch (err) { console.error("Terjadi kesalahan pada sistem:", err); // Fallback aman agar bot tidak crash / layar putih if (typeof appendBotMessageToUI === "function") { appendBotMessageToUI("Maaf bro, terjadi sedikit kendala pada sistem. Coba ulangi pesan kamu ya!"); } } }
