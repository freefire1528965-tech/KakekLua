// Contoh fungsi penanganan pesan masuk pada AI Assistant function handleUserMessage(input) { const cleanInput = input.trim().toLowerCase(); // Daftar kata sapaan atau percakapan umum const greetings = ["halo", "hai", "p", "assalamualaikum", "selamat pagi", "selamat malam", "hi"]; // 1. Jika pengguna mengirim sapaan if (greetings.includes(cleanInput)) { return { type: "chat", message: "Halo! Selamat datang di Asisten KAKEK_VIPP ⚡. Silakan ketik nama game Roblox yang ingin Anda cari skripnya, dan saya akan dengan senang hati membantu menemukannya." }; } // 2. Jika input menanyakan bantuan / fitur else if (cleanInput.includes("tolong") || cleanInput.includes("cara") || cleanInput.includes("fitur")) { return { type: "chat", message: "Tentu! Anda bisa memasukkan nama game simulator atau map Roblox tertentu untuk mendapatkan skrip otomatisasi seperti auto farm atau teleport." }; } // 3. Jika input adalah nama game (Pencarian Skrip) else { // Cek apakah game tersedia di database Anda let scriptData = searchScriptInDatabase(cleanInput); if (scriptData) { return { type: "script", gameName: input, tags: scriptData.tags, // contoh: ["KEYLESS", "AUTO FARM"] loadstring: scriptData.url }; } else { return { type: "chat", message: `Mohon maaf, skrip untuk game "${input}" saat ini belum tersedia di database KAKEK_VIPP. Silakan coba cari nama game lain.` }; } } }
